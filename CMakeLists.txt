cmake_minimum_required(VERSION 3.15)
project(DS_PBL_Expression_Evaluator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(expr_eval src/main.cpp src/Expression.cpp src/Parentheses.cpp)
target_include_directories(expr_eval PRIVATE include)
add_executable(test_runner tests/test_main.cpp src/Expression.cpp src/Parentheses.cpp)
target_include_directories(test_runner PRIVATE include)
enable_testing()
add_test(NAME project_tests COMMAND test_runner)
